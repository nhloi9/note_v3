5. Hàm, closure và giá trị trả về

Hàm là first-class value trong cả ba ngôn ngữ.

TypeScript

TypeScript

}

  return values.reduce((acc, x) => acc + x, 0);

Python

Python

    return a + b

        nonlocal n
        n += 1
        return n

7. Object model: class, struct, interface

TypeScript và Python có class; Go chủ yếu dùng struct + method + interface + composition.

TypeScript

TypeScript
interface Speaker {
  speak(): string;
}

class User implements Speaker {
  constructor(
    public name: string,
    private age: number
  ) {}

  speak(): string {
    return `Hi ${this.name}`;
  }
}

Interface là structural: object có đủ shape thì tương thích, không bắt buộc khai báo implements.

Go

Go
type Speaker interface {
    Speak() string
}

type User struct {
    Name string
    age  int // unexported
}

func (u User) Speak() string {
    return "Hi " + u.Name
}

func (u *User) Birthday() {
    u.age++
}

Không viết implements. Chỉ cần type có đủ method thì tự động thỏa interface.

Python

Python
from dataclasses import dataclass
from typing import Protocol

class Speaker(Protocol):
    def speak(self) -> str: ...

@dataclass
class User:
    name: str
    age: int

    def speak(self) -> str:
        return f"Hi {self.name}"

Protocol tạo structural typing cho type checker, gần với interface của TypeScript/Go.

8. Xử lý lỗi

TypeScript/Python thường dùng exception; Go trả lỗi như một giá trị bình thường.

TypeScript

TypeScript
function parseAge(value: string): number {
  const age = Number(value);
  if (Number.isNaN(age)) {
    throw new Error("Invalid age");
  }
  return age;
}

try {
  parseAge("x");
} catch (error: unknown) {
  if (error instanceof Error) console.error(error.message);
}

Go

Go
func parseAge(value string) (int, error) {
    age, err := strconv.Atoi(value)
    if err != nil {
        return 0, fmt.Errorf("parse age: %w", err)
    }
    return age, nil
}

age, err := parseAge("x")
if err != nil {
    log.Println(err)
}

Python

Python
def parse_age(value: str) -> int:
    try:
        return int(value)
    except ValueError as exc:
        raise ValueError("Invalid age") from exc

try:
    parse_age("x")
except ValueError as exc:
    print(exc)
panic trong Go không phải cách thay thế cho throw trong logic thông thường. Dùng error cho lỗi có thể dự đoán và xử lý.

TypeScript

TypeScript
async function getUser(id: number) {
  const response = await fetch(`/users/${id}`);
  if (!response.ok) throw new Error("HTTP error");
  return response.json();
}

const [a, b] = await Promise.all([
  getUser(1),
  getUser(2)
]);

Go

Go
results := make(chan User, 2)
errors := make(chan error, 2)

for _, id := range []int{1, 2} {
    go func(id int) {
        user, err := getUser(id)
        if err != nil { errors <- err; return }
        results <- user
    }(id)
}

select {
case user := <-results:
    fmt.Println(user)
case err := <-errors:
    log.Println(err)
}

Python

Python
import asyncio
import httpx

async def get_user(user_id: int):
    async with httpx.AsyncClient() as client:
        response = await client.get(f"/users/{user_id}")
        response.raise_for_status()
        return response.json()

users = await asyncio.gather(
    get_user(1),
    get_user(2)
)
I/O-bound: cả ba đều xử lý concurrency tốt. CPU-bound: Node.js cần Worker Threads/process; Python thường dùng multiprocessing/native extension; Go chạy goroutine song song trên nhiều OS thread khi có nhiều CPU core.

11. File và JSON

Đọc file JSON là ví dụ I/O cơ bản, đồng thời cho thấy style của từng ngôn ngữ.

TypeScript

TypeScript
import { readFile, writeFile } from "node:fs/promises";

type Config = { port: number };

const raw = await readFile("config.json", "utf8");
const config = JSON.parse(raw) as Config;

await writeFile(
  "out.json",
  JSON.stringify(config, null, 2)
);

Go

Go
type Config struct {
    Port int `json:"port"`
}

data, err := os.ReadFile("config.json")
if err != nil { log.Fatal(err) }

var config Config
if err := json.Unmarshal(data, &config); err != nil {
    log.Fatal(err)
}

out, _ := json.MarshalIndent(config, "", "  ")
_ = os.WriteFile("out.json", out, 0644)

Python

Python
import json
from pathlib import Path

config = json.loads(
    Path("config.json").read_text(encoding="utf-8")
)

Path("out.json").write_text(
    json.dumps(config, indent=2),
    encoding="utf-8"
)

Go
func main() {
    http.HandleFunc("/hello", func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")

            "message": "Hello",
        })
    })
